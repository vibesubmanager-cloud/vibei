package com.oriente.oriente_native.speech

import android.content.Context
import android.content.Intent
import android.os.Bundle
import android.os.Handler
import android.os.Looper
import android.speech.RecognitionListener
import android.speech.RecognizerIntent
import android.speech.SpeechRecognizer
import android.speech.tts.TextToSpeech
import android.speech.tts.UtteranceProgressListener
import android.util.Log
import java.io.File
import java.util.Locale
import java.util.UUID
import java.util.concurrent.ConcurrentHashMap

/**
 * TTS via Android TextToSpeech.
 * ASR via standard cloud SpeechRecognizer (simple path that worked on device).
 */
class SpeechModule(
    private val context: Context,
    private val onAsr: (text: String, isFinal: Boolean) -> Unit,
    private val onTtsSpeakingChanged: ((Boolean) -> Unit)? = null,
    private val onErrorSpokenReady: (() -> Unit)? = null,
) {
    private val main = Handler(Looper.getMainLooper())
    private var tts: TextToSpeech? = null
    private var ttsReady = false
    private var speechRate = 1.0f
    private var speechVolume = 1.0f

    private var recognizer: SpeechRecognizer? = null
    private var listening = false
    private var wantListening = false
    private var ttsSpeaking = false
    private var taskMode = false
    private var minConfidence = 0.35f
    private val afterSpeak = ConcurrentHashMap<String, () -> Unit>()
    @Volatile private var speakingText: String = ""
    private var ducked = false
    private var restartRunnable: Runnable? = null

    val isTtsReady: Boolean get() = ttsReady
    val isVoskReady: Boolean
        get() = File(context.filesDir, "vosk-model-small-en-us").exists()

    fun init() {
        tts = TextToSpeech(context) { status ->
            ttsReady = status == TextToSpeech.SUCCESS
            if (ttsReady) {
                tts?.language = Locale.getDefault()
                tts?.setSpeechRate(speechRate)
                tts?.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
                    override fun onStart(utteranceId: String?) {
                        ttsSpeaking = true
                        onTtsSpeakingChanged?.invoke(true)
                        if (wantListening && !listening && taskMode) {
                            main.postDelayed({ beginListen() }, 80)
                        }
                    }

                    override fun onDone(utteranceId: String?) {
                        ttsSpeaking = false
                        onTtsSpeakingChanged?.invoke(false)
                        val id = utteranceId ?: return
                        val cb = afterSpeak.remove(id)
                        main.post {
                            cb?.invoke()
                            if (wantListening && !listening) {
                                main.postDelayed({ beginListen() }, 280)
                            }
                        }
                    }

                    @Deprecated("Deprecated in Java")
                    override fun onError(utteranceId: String?) {
                        ttsSpeaking = false
                        onTtsSpeakingChanged?.invoke(false)
                        val id = utteranceId ?: return
                        val cb = afterSpeak.remove(id)
                        main.post {
                            cb?.invoke()
                            if (wantListening && !listening) {
                                main.postDelayed({ beginListen() }, 280)
                            }
                        }
                    }
                })
            }
            onErrorSpokenReady?.invoke()
        }
    }

    fun setSpeechRate(rate: Double) {
        speechRate = rate.toFloat().coerceIn(0.5f, 2.0f)
        tts?.setSpeechRate(speechRate)
    }

    fun setSpeechVolume(volume: Double) {
        speechVolume = volume.toFloat().coerceIn(0.2f, 1.0f)
    }

    fun setTaskMode(enabled: Boolean) {
        taskMode = enabled
    }

    fun setMinConfidence(value: Double) {
        minConfidence = value.toFloat().coerceIn(0.1f, 1.0f)
    }

    fun stopSpeaking() {
        runCatching { tts?.stop() }
        ttsSpeaking = false
        speakingText = ""
        ducked = false
        onTtsSpeakingChanged?.invoke(false)
        afterSpeak.clear()
    }

    fun setSpeakingText(text: String) {
        speakingText = text.lowercase(Locale.getDefault())
    }

    fun duckVolume(duck: Boolean) {
        ducked = duck
    }

    fun speak(text: String, interrupt: Boolean = true) {
        val engine = tts ?: return
        if (!ttsReady) return
        speakingText = text.lowercase(Locale.getDefault())
        val mode = if (interrupt) TextToSpeech.QUEUE_FLUSH else TextToSpeech.QUEUE_ADD
        val vol = if (ducked) (speechVolume * 0.25f).coerceIn(0.05f, 1f) else speechVolume
        val params = Bundle().apply {
            putFloat(TextToSpeech.Engine.KEY_PARAM_VOLUME, vol)
        }
        engine.speak(text, mode, params, UUID.randomUUID().toString())
    }

    fun speakThen(text: String, gapMs: Long = 400, after: () -> Unit) {
        val engine = tts
        if (engine == null || !ttsReady || text.isBlank()) {
            main.postDelayed(after, gapMs)
            return
        }
        val id = UUID.randomUUID().toString()
        afterSpeak[id] = {
            main.postDelayed(after, gapMs)
        }
        val params = Bundle().apply {
            putFloat(TextToSpeech.Engine.KEY_PARAM_VOLUME, speechVolume)
        }
        engine.speak(text, TextToSpeech.QUEUE_FLUSH, params, id)
    }

    fun startListening() {
        main.post {
            wantListening = true
            if (ttsSpeaking && !taskMode) return@post
            if (!SpeechRecognizer.isRecognitionAvailable(context)) {
                speak(
                    "Speech recognition is not available on this phone. " +
                        "Please install or update the Google app and allow microphone.",
                )
                return@post
            }
            // Always reset so a stuck "listening" flag cannot block Tap-and-speak.
            runCatching { recognizer?.cancel() }
            listening = false
            beginListen()
        }
    }

    private fun beginListen() {
        if (!wantListening) return
        if (ttsSpeaking && !taskMode) return
        if (listening) return
        ensureRecognizer()
        listening = true
        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
            putExtra(
                RecognizerIntent.EXTRA_LANGUAGE_MODEL,
                RecognizerIntent.LANGUAGE_MODEL_FREE_FORM,
            )
            putExtra(RecognizerIntent.EXTRA_PARTIAL_RESULTS, true)
            putExtra(RecognizerIntent.EXTRA_MAX_RESULTS, 5)
            putExtra(RecognizerIntent.EXTRA_LANGUAGE, Locale.getDefault().toLanguageTag())
            putExtra(RecognizerIntent.EXTRA_CALLING_PACKAGE, context.packageName)
            putExtra(RecognizerIntent.EXTRA_PREFER_OFFLINE, false)
            putExtra(
                RecognizerIntent.EXTRA_SPEECH_INPUT_COMPLETE_SILENCE_LENGTH_MILLIS,
                if (taskMode) 1500 else 2500,
            )
            putExtra(
                RecognizerIntent.EXTRA_SPEECH_INPUT_POSSIBLY_COMPLETE_SILENCE_LENGTH_MILLIS,
                if (taskMode) 1000 else 1800,
            )
        }
        try {
            Log.i(TAG, "beginListen taskMode=$taskMode")
            recognizer?.startListening(intent)
        } catch (t: Throwable) {
            Log.e(TAG, "startListening failed", t)
            listening = false
            if (!taskMode) {
                speak("Could not start speech recognition. Update the Google app and try again.")
            }
        }
    }

    private fun scheduleRestart(delayMs: Long) {
        if (!wantListening) return
        restartRunnable?.let { main.removeCallbacks(it) }
        val r = Runnable {
            restartRunnable = null
            if (wantListening && !listening) beginListen()
        }
        restartRunnable = r
        main.postDelayed(r, delayMs)
    }

    private fun ensureRecognizer() {
        if (recognizer == null) {
            recognizer = SpeechRecognizer.createSpeechRecognizer(context).apply {
                setRecognitionListener(listener)
            }
        }
    }

    private fun resetRecognizer() {
        runCatching { recognizer?.cancel() }
        runCatching { recognizer?.destroy() }
        recognizer = null
        listening = false
        ensureRecognizer()
    }

    fun stopListening() {
        main.post {
            wantListening = false
            listening = false
            restartRunnable?.let { main.removeCallbacks(it) }
            restartRunnable = null
            runCatching { recognizer?.stopListening() }
            runCatching { recognizer?.cancel() }
        }
    }

    fun release() {
        stopListening()
        afterSpeak.clear()
        recognizer?.destroy()
        recognizer = null
        tts?.shutdown()
        tts = null
        ttsReady = false
    }

    private fun isEchoJunk(text: String): Boolean {
        val t = text.lowercase().trim()
        if (t.isEmpty()) return true
        val junk = listOf(
            "listening",
            "try again",
            "trying again",
            "cancelled",
            "vibei",
            "vivei",
            "speech recognition",
            "i'm still listening",
            "im still listening",
            "still looking",
            "lost it",
        )
        if (junk.any { t == it || t.startsWith("$it ") || (t.contains(it) && t.length < 28) }) {
            return true
        }
        val spoken = speakingText
        if (spoken.isNotEmpty() && t.length >= 3) {
            if (spoken.contains(t) || t.split(" ").all { w -> spoken.contains(w) && w.length > 2 }) {
                val isStop = t.contains("stop") || t.contains("cancel") || t.contains("quiet") ||
                    t.contains("enough") || t.contains("go back")
                if (!isStop) return true
            }
        }
        return false
    }

    private fun passesTaskFilter(text: String): Boolean {
        if (!taskMode) return true
        val allowed = listOf(
            "stop speaking", "be quiet", "mute", "quiet",
            "stop looking", "stop", "cancel", "go back", "back",
            "repeat", "keep looking", "describe", "what is that", "what's that",
            "pause", "resume", "next paragraph", "previous paragraph",
            "next", "previous", "torch", "flashlight",
        )
        val t = text.lowercase()
        return allowed.any { t == it || t.contains(it) }
    }

    private fun bestResult(results: Bundle?): String {
        val texts = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
            ?: return ""
        val scores = results.getFloatArray(SpeechRecognizer.CONFIDENCE_SCORES)
        // Command Tap-and-speak: never drop by confidence — that silently ate commands.
        if (!taskMode || scores == null || scores.isEmpty()) {
            return texts.firstOrNull().orEmpty()
        }
        var bestIdx = 0
        var best = -1f
        for (i in scores.indices) {
            if (scores[i] > best) {
                best = scores[i]
                bestIdx = i
            }
        }
        if (best in 0f..1f && best < minConfidence) return ""
        return texts.getOrNull(bestIdx).orEmpty()
    }

    private val listener = object : RecognitionListener {
        override fun onReadyForSpeech(params: Bundle?) {
            Log.i(TAG, "onReadyForSpeech")
        }

        override fun onBeginningOfSpeech() {
            Log.i(TAG, "onBeginningOfSpeech")
            if (ttsSpeaking) ducked = true
        }

        override fun onRmsChanged(rmsdB: Float) {
            if (ttsSpeaking && rmsdB > 4f) ducked = true
        }

        override fun onBufferReceived(buffer: ByteArray?) {}
        override fun onEndOfSpeech() {}

        override fun onError(error: Int) {
            listening = false
            Log.w(TAG, "ASR error code=$error taskMode=$taskMode want=$wantListening")
            if (!wantListening) return

            val softFail = error == SpeechRecognizer.ERROR_NO_MATCH ||
                error == SpeechRecognizer.ERROR_SPEECH_TIMEOUT
            val retryable = error == SpeechRecognizer.ERROR_CLIENT ||
                error == SpeechRecognizer.ERROR_SERVER ||
                error == SpeechRecognizer.ERROR_NETWORK ||
                error == SpeechRecognizer.ERROR_NETWORK_TIMEOUT ||
                error == SpeechRecognizer.ERROR_RECOGNIZER_BUSY

            if (softFail || retryable || taskMode) {
                if (error == SpeechRecognizer.ERROR_CLIENT ||
                    error == SpeechRecognizer.ERROR_RECOGNIZER_BUSY
                ) {
                    resetRecognizer()
                }
                // Keep Tap-and-speak alive; Flutter owns the 30s silence end.
                scheduleRestart(400)
                return
            }

            wantListening = false
            val msg = when (error) {
                SpeechRecognizer.ERROR_INSUFFICIENT_PERMISSIONS ->
                    "Microphone permission is missing. Allow microphone in settings."
                SpeechRecognizer.ERROR_AUDIO ->
                    "Microphone error. Close other apps using the mic."
                else ->
                    "Speech recognition failed. Update the Google app, allow microphone, then try again."
            }
            speak(msg)
        }

        override fun onResults(results: Bundle?) {
            listening = false
            val text = bestResult(results)
            Log.i(TAG, "onResults text='$text'")
            if (text.isNotEmpty() && !isEchoJunk(text) && passesTaskFilter(text)) {
                onAsr(text, true)
                if (taskMode && wantListening) {
                    scheduleRestart(350)
                } else {
                    wantListening = false
                }
            } else if (wantListening) {
                scheduleRestart(350)
            }
        }

        override fun onPartialResults(partialResults: Bundle?) {
            val text = bestResult(partialResults)
            if (text.isNotEmpty() && !isEchoJunk(text) && passesTaskFilter(text)) {
                Log.i(TAG, "partial='$text'")
                onAsr(text, false)
            }
        }

        override fun onEvent(eventType: Int, params: Bundle?) {}
    }

    companion object {
        private const val TAG = "VibeiSpeech"
    }
}

      tickSound: prefs.getBool('tickSound') ?? true,
      vibration: prefs.getBool('tickVibration') ?? true,
    );
  }

  Future<void> update(PrivacySettings next) async {
    state = next;
    final prefs = await SharedPreferences.getInstance();
    await prefs.setBool('onlineVisionConsent', next.onlineVisionConsent);
    await prefs.setBool('onlineChatConsent', next.onlineChatConsent);
    await prefs.setDouble('speechRate', next.speechRate);
    await prefs.setDouble('speechVolume', next.speechVolume);
    await _host.setPrivacySettings(next);
    await _host.setSpeechRate(next.speechRate);
    await _host.setSpeechVolume(next.speechVolume);
  }
}

class AppController extends StateNotifier<AppUiState> {
  AppController(this._ref) : super(const AppUiState()) {
    _bindStreams();
  }

  final Ref _ref;
  final AppFlowMachine _flow = const AppFlowMachine();
  OrienteHost get _host => _ref.read(orienteHostProvider);

  Timer? _silenceWarn;
  Timer? _silenceEnd;
  StreamSubscription<AsrEvent>? _asrSub;
  DateTime? _listenGraceUntil;
  bool _handlingCommand = false;

  AttemptTrace get _trace => _ref.read(attemptTraceProvider);

  void _publishTrace() {
    _ref.read(attemptTraceProvider.notifier).state = _trace;
  }

  bool get _acceptHomeCommand {
    if (state.listening || state.phase == UiPhase.listening) return true;
    final g = _listenGraceUntil;
    return g != null && DateTime.now().isBefore(g);
  }

  void _bindStreams() {
    _asrSub = _host.asrEvents.listen((event) async {
      final text = event.text.trim();
      final t = _trace;
      t.add(
        'kotlin_to_dart',
        text.isNotEmpty || !event.isFinal,
        'final=${event.isFinal} text="$text"',
      );
      if (!event.isFinal && text.isNotEmpty) {
        t.rawPartial = text;
      }
      if (event.isFinal && text.isNotEmpty) {
        t.rawFinal = text;
      }
      _publishTrace();

      final executing = state.phase == UiPhase.objectSearch ||
          state.phase == UiPhase.textSearch ||
          state.phase == UiPhase.nearbyText ||
          state.phase == UiPhase.nearbyObjects ||
          state.phase == UiPhase.personalRemember ||
          state.phase == UiPhase.documentRead ||
          state.phase == UiPhase.describe;

      if (executing) {
        if (event.isFinal && text.isNotEmpty) {
          await _handleTaskUtterance(text);
        }
        return;
      }

      // Grace: accept finals briefly after listening flag clears (early end bug).
      if (!_acceptHomeCommand) {
        t.add('dart_listener', false, 'dropped: not listening phase=${state.phase}');
        _publishTrace();
        return;
      }
      t.phaseAtReceive = '${state.phase}/listening=${state.listening}';
      t.add('dart_listener', true, t.phaseAtReceive);
      _publishTrace();

      _resetSilenceTimers();
      if (text.isEmpty) return;
      state = state.copyWith(partialTranscript: text);
      if (event.isFinal) {
        await _handleUtterance(text);
      }
    });

    _host.guidance.listen((g) {
      if (g.announce != null && g.announce!.isNotEmpty) {
        state = state.copyWith(statusLine: g.announce);
      } else if (g.directionSpeech != null) {
        state = state.copyWith(statusLine: g.directionSpeech);
      }
      state = state.copyWith(proximity: g.proximity, guidancePhase: g.phase);
    });

    _host.sessions.listen((s) {

  Future<void> _handleUtterance(String text) async {
    if (_handlingCommand) return;
    _handlingCommand = true;
    try {
      _silenceWarn?.cancel();
      _silenceEnd?.cancel();
      await _host.stopListening();
      final heard = text.trim();
      // Keep a short grace so a delayed duplicate final can still land.
      _listenGraceUntil = DateTime.now().add(const Duration(seconds: 2));
      state = state.copyWith(
        listening: false,
        partialTranscript: heard,
        statusLine: 'Heard: $heard',
      );

      const parser = IntentParser();
      var intent = parser.parse(heard);
      final t = _trace;
      t.normalized = heard.toLowerCase();
      t.intent = intent.runtimeType.toString();
      t.target = switch (intent) {
        ObjectSearchIntent(:final query) => query,
        TextSearchIntent(:final query) => query,
        HybridFindIntent(:final objectQuery) => objectQuery,
        _ => '',
      };
      t.add('parsed', intent is! ClarifyIntent, '${t.intent} target=${t.target}');
      _publishTrace();
      _logCommand(heard, t.intent);

      if (intent is ClarifyIntent) {
        // Incomplete ASR final — keep listening; do not bounce to HOME.
        t.flowTransition = 'clarify→stay listening';
        _publishTrace();
        state = state.copyWith(
          phase: UiPhase.listening,
          flow: AppFlowState.listening,
          listening: true,
          statusLine: 'Listening…',
        );
        await _host.startListening(taskMode: false);
        _resetSilenceTimers();
        return;
      }

      if (intent is UnsupportedIntent) {
        final msg = intent.message.trim().isNotEmpty ? intent.message : _unknownMsg;
        t.flowTransition = 'unsupported→home';
        _publishTrace();
        state = state.copyWith(
          phase: UiPhase.home,
          flow: AppFlowState.home,
          listening: false,
          statusLine: 'Ready',
        );
        await _host.speak(msg);
        return;
      }

      // Ensure state machine is in listening before startSearch.
      if (state.flow != AppFlowState.listening) {
        state = state.copyWith(flow: AppFlowState.listening);
      }
      await _dispatch(intent);
    } finally {
      _handlingCommand = false;
    }
  }

  Future<void> _handleTaskUtterance(String text) async {
    const parser = TaskCommandParser();
    final readCmd = parser.parseReadControl(text);
    if (readCmd != null && state.phase == UiPhase.documentRead) {
      state = state.copyWith(readCommand: readCmd);
      return;

    final intent = parser.parse(text);
    if (intent == null) return;

        state = state.copyWith(speechMuted: true);

      case StopLookingIntent():
      case StopIntent():
      case GoBackIntent():
        await cancel();
      case RepeatIntent():

        if (last != null && last.isNotEmpty) {
          await _host.speak(last, interrupt: true);

      case KeepLookingIntent():

          statusLine: 'Still looking for ${state.activeQuery ?? 'it'}',

  }

  AppIntent parse(String raw) {
    var text = _normalize(raw);
    if (text.isEmpty) {
      return const ClarifyIntent('I did not catch that. Please say it again.');
    }

    text = _stripFillers(text);

    if (_matchesAny(text, const [
      'stop speaking',
      'be quiet',
      'mute',
    ])) {
      return const StopSpeakingIntent();
    }
    if (_matchesAny(text, const ['stop looking', 'stop searching'])) {
      return const StopLookingIntent();
    }
    if (_matchesAny(text, const ['keep looking', 'keep searching'])) {
      return const KeepLookingIntent();
    }
    if (_matchesAny(text, const ['repeat', 'say again'])) {
      return const RepeatIntent();
    }

    // Bare stop — but not "stop sign" / find-stop targets.
    if (_matchesAny(text, _stopPhrases) &&
        !text.contains('stop sign') &&
        !RegExp(r'\b(find|look|search|where)\b').hasMatch(text)) {
      return const StopIntent();
    }
    if (_matchesAny(text, const ['go back', 'back', 'home'])) {
      return const GoBackIntent();

      'text saying',
    ])) {

            'look for the sign that says',
            'find the sign that says',
            'sign that says',
            'find text that says',
            'find text saying',
            'text that says',
            'text saying',
            'find the word',
            'find the text',
            'look for the word',
            'search for the word',
            'find word',
            'search for text',
            'look for text',
            'search text',
            'find text',
          ]) ??
          '';
      if (q.trim().isEmpty) {
        return const ClarifyIntent('Which word should I find?');
      }
      return TextSearchIntent(q.trim());
    }

    if (_containsAny(text, const [
      'describe in detail',
      'describe more clearly',
      'give me a full description',
      'full description',
      'more detail',
    ])) {
      return const DescribeDetailIntent();
    }

    if (_containsAny(text, const [
      'what is that',
      "what's that",
      'whats that',
      'identify this',
      'identify that',
      'what am i looking at',
    ])) {
      return const IdentifyIntent();
    }

  String _normalize(String raw) {
    return raw
        .toLowerCase()
        .replaceAll("'", '') // where's → wheres (matched by wheres prefixes)
        .replaceAll(RegExp(r'[^a-z0-9\s]'), ' ')
        .replaceAll(RegExp(r'\s+'), ' ')
        .trim();
  }

  String _stripFillers(String text) {
    var t = text;
    for (var i = 0; i < 6; i++) {
      final next = t
          .replaceFirst(
            RegExp(
              r'^(hello|hi|hey|okay|ok|please|excuse me|thanks|thank you)[, ]+',
            ),
            '',
          )
          .replaceFirst(
            RegExp(
              r'^(can you|could you|would you|will you|can u)\s+',
            ),
            '',
          )
          .replaceFirst(
            RegExp(
              r'^(i want to|i need to|help me|i wanna)\s+',
            ),
            '',
          )
          .replaceFirst(RegExp(r'\s+for me$'), '')
          .replaceFirst(RegExp(r'\s+please$'), '')
          .replaceFirst(RegExp(r'\s+thanks$'), '')
          .trim();
      if (next == t) break;
      t = next;
    }
    return t;
  }

  String _stripArticles(String q) {
    return q
        .replaceFirst(RegExp(r'^(a|an|the|my|this|that)\s+'), '')
        .replaceFirst(RegExp(r'^(a|an|the|my|this|that)\s+'), '')
        .replaceFirst(RegExp(r'\s+(is|are|was|were)$'), '')
        .trim();
  }

  bool _hasNoun(String q) {
    final parts = q.split(' ').where((e) => e.length > 1).toList();
    return parts.isNotEmpty;
  }

  bool _matchesAny(String text, List<String> phrases) {
    for (final p in phrases) {
      if (text == p || fuzzyEquals(text, p, threshold: 0.88)) return true;
    }
    return false;
  }

  bool _containsAny(String text, List<String> phrases) {
    for (final p in phrases) {
      if (text.contains(p)) return true;
      if (bestFuzzyScore(p, text) >= 0.9) return true;
    }
    return false;
  }

  bool _startsWithAny(String text, List<String> prefixes) {
    for (final p in prefixes) {
      if (text == p || text.startsWith('$p ')) return true;
    }
    return false;
  }

  String? _extractAfter(String text, List<String> prefixes) {
    // Longest prefix first.
    final sorted = [...prefixes]..sort((a, b) => b.length.compareTo(a.length));
    for (final p in sorted) {
      if (text == p) return '';
      if (text.startsWith('$p ')) {
        return text.substring(p.length).trim();
      }
      if (text.startsWith(p) && text.length > p.length) {
        return text.substring(p.length).trim();
      }
      final idx = text.indexOf(' $p ');
      if (idx >= 0) {
        return text.substring(idx + p.length + 2).trim();
      }
      if (text.endsWith(' $p')) return '';
    }
    return null;
  }

  bool _looksConversational(String text) {
    const starters = [
      'why',
      'how',
      'who',
      'when',
      'please tell',
      'explain',
      'tell me about',
      'what do you think',
      'help me understand',
    ];
    for (final s in starters) {
      if (text.startsWith(s)) return true;
    }
    final words = text.split(' ');
    return words.length >= 10;
  }

  HybridFindIntent? _hybridFind(String q) {
    final n = q.toLowerCase().trim();
    if (n == 'stop sign' ||
        n == 'a stop sign' ||
        n == 'the stop sign' ||
        n == 'stop signs') {
      return const HybridFindIntent(
        objectQuery: 'stop sign',
        textQuery: 'stop',
      );
    }
    return null;
  }
}
